// Estructura de Datos en Memoria (Arreglo de Tareas)
let tasks = [];

// Selección de Elementos del DOM
const taskForm = document.getElementById('task-form');
const taskNameInput = document.getElementById('task-name');
const courseNameInput = document.getElementById('course-name');
const dueDateInput = document.getElementById('due-date');
const priorityInput = document.getElementById('priority');
const formAlert = document.getElementById('form-alert');

const tasksContainer = document.getElementById('tasks-container');
const filterPriority = document.getElementById('filter-priority');
const filterStatus = document.getElementById('filter-status');

// Elementos de Estadísticas
const statTotal = document.getElementById('stat-total');
const statPending = document.getElementById('stat-pending');
const statCompleted = document.getElementById('stat-completed');
const statOverdue = document.getElementById('stat-overdue');
const progressBar = document.getElementById('progress-bar');

// Inicialización de la Aplicación
document.addEventListener('DOMContentLoaded', () => {
    loadTasksFromStorage();
    renderTasks();
});

// Evento: Formulario de Registro de Tarea
taskForm.addEventListener('submit', (e) => {
    e.preventDefault();
    
    // Captura y Sanitización de Datos
    const name = taskNameInput.value.trim();
    const course = courseNameInput.value.trim();
    const dueDate = dueDateInput.value;
    const priority = priorityInput.value;

    // ALGORITMO 1: Validación de Datos de Entrada
    if (!validateInputs(name, course, dueDate)) {
        showAlert('Por favor, complete todos los campos obligatorios.', 'error');
        return;
    }

    // Creación del objeto Tarea
    const newTask = {
        id: Date.now().toString(),
        name: name,
        course: course,
        dueDate: dueDate,
        priority: priority,
        completed: false,
        createdAt: new Date().toISOString()
    };

    // Agregar al arreglo
    tasks.push(newTask);
    saveTasksToStorage();
    
    // Resetear formulario y actualizar UI
    taskForm.reset();
    showAlert('¡Tarea registrada con éxito!', 'success');
    renderTasks();
});

// ALGORITMO 1 (Auxiliar): Función de Validación
function validateInputs(name, course, dueDate) {
    if (name === '' || course === '' || dueDate === '') {
        return false;
    }
    return true;
}

// ALGORITMO 2: Ordenamiento Cronológico (Inserción/Burbuja por fecha)
function sortTasksByDate(taskList) {
    return taskList.sort((a, b) => {
        const dateA = new Date(a.dueDate);
        const dateB = new Date(b.dueDate);
        return dateA - dateB; // Orden ascendente: más cercana primero
    });
}

// ALGORITMO 3: Filtrado Multicriterio de Tareas
function getFilteredTasks() {
    const selectedPriority = filterPriority.value;
    const selectedStatus = filterStatus.value;

    return tasks.filter(task => {
        const matchesPriority = (selectedPriority === 'TODAS') || (task.priority === selectedPriority);
        let matchesStatus = true;

        if (selectedStatus === 'PENDIENTE') {
            matchesStatus = !task.completed;
        } else if (selectedStatus === 'COMPLETADA') {
            matchesStatus = task.completed;
        }

        return matchesPriority && matchesStatus;
    });
}

// ALGORITMO 4: Cálculo e Interpretación de Estadísticas
function updateStatistics() {
    const total = tasks.length;
    const completed = tasks.filter(t => t.completed).length;
    const pending = total - completed;
    
    const today = new Date().toISOString().split('T')[0];
    const overdue = tasks.filter(t => !t.completed && t.dueDate < today).length;

    const percentage = total === 0 ? 0 : Math.round((completed / total) * 100);

    // Renderizar en UI
    statTotal.textContent = total;
    statPending.textContent = pending;
    statCompleted.textContent = completed;
    statOverdue.textContent = overdue;

    progressBar.style.width = `${percentage}%`;
    progressBar.textContent = `${percentage}%`;
}

// Función Principal de Renderizado de Interfaz
function renderTasks() {
    // 1. Filtrar
    let filtered = getFilteredTasks();
    // 2. Ordenar cronológicamente
    filtered = sortTasksByDate(filtered);

    tasksContainer.innerHTML = '';

    if (filtered.length === 0) {
        tasksContainer.innerHTML = `<p class="empty-msg">No hay tareas que coincidan con los criterios seleccionados.</p>`;
    } else {
        filtered.forEach(task => {
            const taskCard = document.createElement('div');
            taskCard.className = `task-item priority-${task.priority} ${task.completed ? 'completed' : ''}`;
            
            taskCard.innerHTML = `
                <div class="task-info">
                    <h3 class="task-title">${escapeHTML(task.name)}</h3>
                    <p class="task-meta">📖 <strong>Curso:</strong> ${escapeHTML(task.course)} | 📅 <strong>Entrega:</strong> ${task.dueDate} | 🏷️ <strong>Prioridad:</strong> ${task.priority}</p>
                </div>
                <div class="task-actions">
                    <button onclick="toggleTaskStatus('${task.id}')" class="btn-sm btn-complete">
                        ${task.completed ? '↩️ Reactivar' : '✅ Completar'}
                    </button>
                    <button onclick="deleteTask('${task.id}')" class="btn-sm btn-delete">🗑️ Eliminar</button>
                </div>
            `;
            tasksContainer.appendChild(taskCard);
        });
    }

    // 3. Actualizar panel de estadísticas
    updateStatistics();
}

// Operaciones CRUD adicionales
function toggleTaskStatus(id) {
    tasks = tasks.map(t => {
        if (t.id === id) {
            return { ...t, completed: !t.completed };
        }
        return t;
    });
    saveTasksToStorage();
    renderTasks();
}

function deleteTask(id) {
    if (confirm('¿Está seguro de que desea eliminar esta tarea académica?')) {
        tasks = tasks.filter(t => t.id !== id);
        saveTasksToStorage();
        renderTasks();
    }
}

// Persistencia en localStorage
function saveTasksToStorage() {
    localStorage.setItem('uniplan_tasks', JSON.stringify(tasks));
}

function loadTasksFromStorage() {
    const stored = localStorage.getItem('uniplan_tasks');
    if (stored) {
        try {
            tasks = JSON.parse(stored);
        } catch (e) {
            tasks = [];
        }
    }
}

// Escuchadores de eventos para los filtros
filterPriority.addEventListener('change', renderTasks);
filterStatus.addEventListener('change', renderTasks);

// Utilidades Auxiliares
function showAlert(msg, type) {
    formAlert.textContent = msg;
    formAlert.className = `alert alert-${type}`;
    setTimeout(() => {
        formAlert.className = 'alert hidden';
    }, 3000);
}

function escapeHTML(str) {
    return str.replace(/[&<>'"]/g, 
        tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)
    );
}
